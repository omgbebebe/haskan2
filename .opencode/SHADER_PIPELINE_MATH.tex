\section{Coordinate Systems}

\subsection{World Space}
\begin{itemize}
\item Right-handed coordinate system
\item +X = right, +Y = up, +Z = forward (toward viewer in default view)
\item Matches glTF 2.0 convention exactly
\item All procedural meshes (cube, sphere, plane) have outward-facing normals and CCW winding when viewed from outside
\item glTF models load with standard CCW winding
\end{itemize}

\subsection{View Space (Custom lookAt)}
\begin{itemize}
\item \textbf{Right-handed} coordinate system (det = +1)
\item Forward: $-Z$ (points from camera toward scene)
\item Right: $+X$ = world right
\item Up: $+Y$ = world up
\item View matrix $\mathbf{V}$ transforms world $\to$ view
\item linear stores matrices in \textbf{row-major} order
\end{itemize}

\textbf{lookAt construction} (Camera.hs, \texttt{lookAtNegativeYUp} with $\mathbf{u} = (0,1,0)$):
\begin{align}
\mathbf{f} &= \text{normalize}(\mathbf{c} - \mathbf{e}) \quad \text{(forward, points toward target)} \\
\mathbf{s} &= \text{normalize}(\mathbf{f} \times \mathbf{u}) \quad \text{(right)} \\
\mathbf{u'} &= \mathbf{s} \times \mathbf{f} \quad \text{(true up)}
\end{align}

View matrix (row-major):
\[
\mathbf{V} = \begin{pmatrix}
s_x & s_y & s_z & -\mathbf{s} \cdot \mathbf{e} \\
u'_x & u'_y & u'_z & -\mathbf{u'} \cdot \mathbf{e} \\
-f_x & -f_y & -f_z & \mathbf{f} \cdot \mathbf{e} \\
0 & 0 & 0 & 1
\end{pmatrix}
\]

\textbf{Key point}: The third row is $-\mathbf{f}$ because in the view space, $-Z$ is forward (toward the scene). This is a right-handed convention.

\textbf{Verification}: With camera at $(0,10,17)$ looking at origin:
\begin{itemize}
\item $\mathbf{f} \approx (0, -0.5, -0.87)$
\item $\mathbf{s} = \mathbf{f} \times (0,1,0) = (1, 0, 0)$ \checkmark
\item $\mathbf{u'} = \mathbf{s} \times \mathbf{f} = (0, 0.87, -0.5)$ \checkmark
\item $\det(\mathbf{V}_{3\times3}) = +1$ \checkmark
\end{itemize}

\subsection{Clip Space}
\begin{itemize}
\item After projection: $x, y, z, w$ in homogeneous coordinates
\item Vulkan NDC: $x \in [-1, 1]$ (left to right), $y \in [-1, 1]$ (bottom to top)
\item After perspective divide: $x_{ndc} = x/w$, $y_{ndc} = y/w$, $z_{ndc} = z/w$
\end{itemize}

\subsection{Framebuffer Space}
\begin{itemize}
\item Origin $(0, 0)$ at \textbf{top-left}
\item $y$ increases \textbf{downward}
\item Viewport transform uses \textbf{negative height} to map NDC $+Y$ to screen top:
\end{itemize}

\begin{verbatim}
VkViewport viewport = {
    .x = 0,
    .y = height,           // Start from bottom
    .width = width,
    .height = -height,     // Negative: NDC +Y -> screen top
    .minDepth = 0.0,
    .maxDepth = 1.0
};
\end{verbatim}

This gives:
\begin{itemize}
\item NDC $y = +1$ $\to$ screen $y = 0$ (top) --- world +Y at screen top \checkmark
\item NDC $y = -1$ $\to$ screen $y = h$ (bottom)
\item Winding is preserved (negative viewport does not change CCW/CW)
\end{itemize}

\section{Vertex Transformation Pipeline}

\subsection{Step 1: Model to World}
Model matrix $\mathbf{M}$ from entity transform (TRS decomposition).
\[
\mathbf{p}_{world} = \mathbf{M} \cdot \mathbf{p}_{model}
\]

\subsection{Step 2: World to View}
View matrix $\mathbf{V}$ from \texttt{lookAtNegativeYUp} with $\mathbf{u} = (0,1,0)$:
\[
\mathbf{p}_{view} = \mathbf{V} \cdot \mathbf{p}_{world}
\]

\subsection{Step 3: View to Clip --- Standard Projection}
Projection matrix $\mathbf{P}$ uses standard \texttt{Linear.Projection.perspective} \textbf{without} Y-flip:
\[
\mathbf{P} = \mathbf{P}_{persp}
\]

The Y-down-to-screen behavior is achieved via \textbf{negative viewport height} only.

\subsection{Step 4: Perspective Divide}
\[
\mathbf{p}_{ndc} = \begin{pmatrix} x_{clip}/w_{clip} \\ y_{clip}/w_{clip} \\ z_{clip}/w_{clip} \end{pmatrix}
\]

\subsection{Step 5: Viewport Transform (Negative Height)}
\[
x_{screen} = 0 + \frac{x_{ndc} + 1}{2} \cdot width, \quad
y_{screen} = h + \frac{y_{ndc} + 1}{2} \cdot (-h) = h - \frac{(y_{ndc} + 1) \cdot h}{2}
\]

\section{Full Chain: World +Y to Screen}

World point $(0, 1, 0)$:
\begin{enumerate}
\item \textbf{View}: $\mathbf{V}$ maps world $+Y$ to view $+Y$ (positive, since $\mathbf{u'} = \mathbf{s} \times \mathbf{f}$ preserves up direction)
\item \textbf{Projection}: $\mathbf{P}$ maps view $+Y$ to NDC $+y$ ($f_y > 0$ in standard perspective)
\item \textbf{Viewport}: negative height maps NDC $y = +1$ to screen $y = 0$ (top)
\end{enumerate}

Result: World $+Y$ appears at screen \textbf{top}. Correct for Y-up world matching glTF.

\section{Camera Orbital Math}

\subsection{worldUp Definition}
\begin{verbatim}
worldUp :: V3 CFloat
worldUp = V3 0 1 0
\end{verbatim}

Used consistently for:
\begin{itemize}
\item Azimuth (yaw) rotation axis: \texttt{axisAngle worldUp (-az)}
\item Right vector computation in rotation handler: \texttt{worldUp `cross` fwd}
\item lookAt up vector in \texttt{orbitalToMatrix}
\end{itemize}

\subsection{Azimuth and Elevation}
\begin{itemize}
\item Azimuth: Rotation around worldUp (yaw). Positive = counter-clockwise viewed from above.
\item Elevation: Rotation around local right axis. \texttt{axisAngle right el} --- positive elevation = camera above target looking down.
\item Default camera: azimuth = 0\textdegree, elevation = 30\textdegree (above horizon)
\end{itemize}

\subsection{Camera Position}
Camera offset starts at $\text{V3 } 0\ 0\ d$ and is rotated by orientation quaternion:
\[
\mathbf{p}_{camera} = \mathbf{target} + q \cdot (0, 0, d)
\]

At 30\textdegree elevation: $q$ rotates $(0,0,d)$ to $(0, d \sin 30\textdegree, d \cos 30\textdegree) = (0, d/2, d\sqrt{3}/2)$.

Camera at $Y = +d/2$, above target. \checkmark

\section{UV Conventions}

\subsection{Vulkan Texture Sampling}
\begin{itemize}
\item UV $(0,0)$ = \textbf{top-left} of texture image
\item UV $(1,1)$ = \textbf{bottom-right} of texture image
\item V increases \textbf{downward} through the image
\item Row 0 of uploaded pixel data = top of texture (JuicyPixels row 0 = top, matching Vulkan)
\end{itemize}

\subsection{Procedural Meshes (unitCube)}
Side faces: bottom vertices ($Y = -0.5$) get $V = 1$ (bottom of texture), top vertices ($Y = +0.5$) get $V = 0$ (top of texture). This places arrow-tip (top of image) at top of face, arrows pointing $+Y$.

\begin{tabular}{lll}
\toprule
Face & V convention & Notes \\
\midrule
Front/Back/Left/Right & $V=0$ at $Y=+s$, $V=1$ at $Y=-s$ & Arrows point $+Y$ \\
Top ($+Y$) & $V=0$ at $Z=-s$, $V=1$ at $Z=+s$ & Correct, unchanged \\
Bottom ($-Y$) & $V=0$ at $Z=+s$, $V=1$ at $Z=-s$ & Winding: CCW from $-Y$ \\
\bottomrule
\end{tabular}

\subsection{Procedural Meshes (uvSphere)}
\begin{itemize}
\item $U = 1 - \text{lonIdx}/\text{lonCount}$ --- U increases clockwise viewed from $+Y$, so front-facing polygons sample U left-to-right
\item $V = \text{latIdx}/\text{latCount}$ --- $V=0$ at north pole ($+Y$), $V=1$ at south pole ($-Y$)
\item Tangent: $(\sin\phi, 0, -\cos\phi)$ --- matches inverted U direction
\end{itemize}

\subsection{glTF Models}
\begin{itemize}
\item glTF spec says UV $(0,0)$ = top-left, but most exporters (including Blender) write UVs in OpenGL convention where $V=0$ is at the \textbf{bottom} of the face
\item \textbf{flipV applied}: $V' = 1 - V$ in \texttt{primitiveToVertices} (GLTF.hs)
\item Tangent handedness negated ($W' = -W$) to match V flip
\item Tangents computed from \textbf{flipped} UVs (not original) to maintain consistency
\end{itemize}

\section{Skybox Ray Computation}

\subsection{Current Implementation}
\begin{verbatim}
computeSkyboxRays view proj =
  let V4 (V4 v00 v01 v02 _) (V4 v10 v11 v12 _) (V4 v20 v21 v22 _) _ = view
      worldRot = V3 (V3 v00 v01 v02) (V3 v10 v11 v12) (V3 v20 v21 v22)
      
      V4 (V4 fx _ _ _) (V4 _ fy _ _) _ _ = proj
      
      ndcToDir (V4 x y _z _w) =
        let viewDir = V3 (x / fx) (y / fy) (-1)
        in normalize (worldRot !* viewDir)
\end{verbatim}

\subsection{Analysis}
\texttt{view} is $\mathbf{V}^T$ (transposed view matrix, column-major for GPU). Rows of $\mathbf{V}^T$ = columns of $\mathbf{V}$:
\begin{itemize}
\item Row 0 of $\mathbf{V}^T$ = column 0 of $\mathbf{V}$ = $(s_x, u'_x, -f_x)$ = right component
\end{itemize}

Wait --- $\mathbf{V}^T$ rows are $\mathbf{V}$ columns. But $\mathbf{V}$ rows are $(\mathbf{s}, \mathbf{u'}, -\mathbf{f})$. So $\mathbf{V}^T$ rows $= \mathbf{V}$ columns $= (\mathbf{s}^T, \mathbf{u'}^T, (-\mathbf{f})^T)$ as column vectors written as rows.

Actually: extracting rows of $\mathbf{V}^T$ gives the same vectors as the columns of $\mathbf{V}$. But in the code, \texttt{worldRot} is formed from the \textbf{rows} of \texttt{view}, where \texttt{view} $= \mathbf{V}^T$. So:

$\text{worldRot}_{ij} = V^T_{ij} = V_{ji}$

This means $\text{worldRot}$ as a matrix acting on column vectors gives:
\[
\text{worldRot} \cdot \mathbf{v} = \mathbf{V}^T \cdot \mathbf{v}
\]

But the rotation part of $\mathbf{V}$ is $\mathbf{R}$ (world$\to$view), and $\mathbf{R}^T = \mathbf{R}^{-1}$ (view$\to$world). Since $\mathbf{V}^T = \mathbf{R}^T$ for the rotation part, $\text{worldRot} = \mathbf{R}^T = \mathbf{R}^{-1}$, which transforms view$\to$world. \checkmark

\subsection{Verification}
At screen center $(x, y) = (0, 0)$:
\[
\mathbf{v}_{view} = (0, 0, -1), \quad \mathbf{v}_{world} = \text{normalize}(\mathbf{R}^{-1} \cdot (0,0,-1)) = \text{forward direction} \quad \checkmark
\]

At screen top $(x, y) = (0, 1)$:
\[
\mathbf{v}_{view} = (0, 1/f_y, -1), \quad \mathbf{v}_{world} \propto \mathbf{u'}/f_y - \mathbf{f}
\]
Points slightly upward from forward. With negative viewport, screen top = NDC $y=+1$ = world up. \checkmark

\section{Resolved Issues}

\subsection{Issue: Reflected View Matrix (Y-down worldUp)}
\textbf{Root cause}: \texttt{worldUp = V3 0 (-1) 0} made $\mathbf{s} = \mathbf{f} \times (0,-1,0)$ produce a negated right vector, giving det $= -1$ (reflection). This reversed winding: CCW$\to$CW, so front faces were culled.

\textbf{Fix}: Changed to \texttt{worldUp = V3 0 1 0}. View matrix now has det $= +1$, preserving CCW winding. glTF models render correctly.

\textbf{Secondary effect}: The reflected view matrix previously compensated for missing \texttt{flipV} in the glTF loader. After fixing worldUp, \texttt{flipV} was restored.

\subsection{Issue: glTF UV V-axis}
\textbf{Root cause}: Blender's glTF exporter writes UVs in OpenGL convention ($V=0$ at bottom of face), but Vulkan has $V=0$ at top of texture.

\textbf{Fix}: Restored \texttt{flipV} ($V' = 1 - V$) in \texttt{primitiveToVertices}. Tangents computed from flipped UVs; handedness negated.

\subsection{Issue: Procedural Cube Side Faces V-inverted}
\textbf{Root cause}: Side face UVs had $V=0$ at bottom ($Y=-s$) and $V=1$ at top ($Y=+s$), but Vulkan $V=0$ = top of texture. So texture top appeared at face bottom.

\textbf{Fix}: Swapped V on all 4 side faces: bottom vertices get $V=1$, top vertices get $V=0$.

\subsection{Issue: Procedural Cube Bottom Face Invisible}
\textbf{Root cause}: Winding produced normal $(0,+1,0)$ instead of $(0,-1,0)$. Face was back-facing.

\textbf{Fix}: Reversed bottom face indices from \texttt{20,21,22, 20,22,23} to \texttt{20,22,21, 20,23,22}.

\subsection{Issue: Procedural Sphere U-mirrored}
\textbf{Root cause}: $U = \text{lonIdx}/\text{lonCount}$ increased counter-clockwise from $+Y$, making U go right-to-left from front view.

\textbf{Fix}: $U = 1 - \text{lonIdx}/\text{lonCount}$. Tangent changed to $(\sin\phi, 0, -\cos\phi)$.

\subsection{Issue: Procedural Sphere V-inverted}
\textbf{Root cause}: $V = 1 - \text{latIdx}/\text{latCount}$ placed $V=0$ at south pole instead of north pole.

\textbf{Fix}: $V = \text{latIdx}/\text{latCount}$.

\section{Critical Invariants}

\begin{enumerate}
\item View matrix must have det $= +1$ (right-handed, no reflection)
\item All meshes (procedural and imported) use CCW winding with outward normals
\item Projection matrix is standard (no Y-flip)
\item Viewport uses negative height for Vulkan Y-down
\item Backface culling enabled with CCW front face
\item Camera up vector: \texttt{worldUp = V3 0 1 0} (used consistently in orbit and lookAt)
\item glTF UVs: \texttt{flipV} applied ($V' = 1-V$), tangents from flipped UVs, handedness negated
\item Procedural mesh UVs: direct Vulkan convention ($V=0$ = texture top at face top)
\end{enumerate}
