\section{Notation and Conventions}
%=============================================================================

\begin{itemize}
    \item Scalars: $a, b, \alpha, \theta$ (italic)
    \item Vectors: $\vv{v}, \vv{n}, \vv{p}$ (bold, lowercase)
    \item Matrices: $\mM{M}, \mM{V}, \mM{P}$ (bold, uppercase)
    \item Homogeneous coordinates: $\vv{p}_h = (p_x, p_y, p_z, 1)^\top$
    \item Direction vectors in homogeneous: $\vv{d}_h = (d_x, d_y, d_z, 0)^\top$
    \item Quaternion: $\vv{q} = (w, x, y, z)$ with scalar part $w$
    \item Column-vector convention: $\vv{v}' = \mM{M} \vv{v}$
    \item Handedness: Right-handed throughout, except Vulkan NDC which is left-handed in $Y$
\end{itemize}

\noindent
\textbf{Critical inconsistency:} The projection matrix produces OpenGL-style clip space ($Y$ up, $Z \in [-1,1]$), but Vulkan NDC requires $Y$ down and $Z \in [0,1]$. \section{CPU-Side Transformations}

%-----------------------------------------------------------------------------
\subsection{Model Matrix}
%-----------------------------------------------------------------------------

File: \texttt{src/Graphics/Haskan/Scene/Transform.hs:24--35}

For an entity with position $\vv{t}$, rotation quaternion $\vv{q}$, and scale $\vv{s} = (s_x, s_y, s_z)$:
\begin{align}
    \mM{S} &= \begin{pmatrix}
        s_x & 0   & 0   & 0 \\
        0   & s_y & 0   & 0 \\
        0   & 0   & s_z & 0 \\
        0   & 0   & 0   & 1
    \end{pmatrix} \\
    \mM{R} &= \text{rotation matrix from } \vv{q} \\
    \mM{T} &= \begin{pmatrix}
        1 & 0 & 0 & t_x \\
        0 & 1 & 0 & t_y \\
        0 & 0 & 1 & t_z \\
        0 & 0 & 0 & 1
    \end{pmatrix} \\
    \mM{model} &= \mM{T} \cdot \mM{R} \cdot \mM{S}
\end{align}

Hierarchy: for a child node with local transform $\mM{local}$ and parent world transform $\mM{parent}$:
\begin{equation}
    \mM{world} = \mM{local} \cdot \mM{parent}
\end{equation}

%-----------------------------------------------------------------------------
\subsection{View Matrix}
%-----------------------------------------------------------------------------

File: \texttt{src/Graphics/Haskan/Camera.hs:93--96}

Using \texttt{linear}'s \texttt{lookAt} with eye $\vv{e}$, center $\vv{c}$, up $\vv{u}$:
\begin{align}
    \vv{z} &= \normalize(\vv{c} - \vv{e}) \\
    \vv{x} &= \normalize(\vv{z} \times \vv{u}) \\
    \vv{y} &= \vv{x} \times \vv{z} \\
    \mM{view} &= \begin{pmatrix}
        x_x & x_y & x_z & -\vv{x} \cdot \vv{e} \\
        y_x & y_y & y_z & -\vv{y} \cdot \vv{e} \\
        -z_x & -z_y & -z_z & \vv{z} \cdot \vv{e} \\
        0 & 0 & 0 & 1
    \end{pmatrix}
\end{align}

%-----------------------------------------------------------------------------
\subsection{Projection Matrix}
%-----------------------------------------------------------------------------

File: \texttt{src/Graphics/Haskan/Engine.hs:1355--1358}

Using \texttt{linear}'s \texttt{perspective} with:
\begin{itemize}
    \item FOV$_y = \pi/12$ (15\degree)
    \item Aspect $a = w/h$
    \item Near $n = 0.1$, Far $f = 10000.0$
\end{itemize}

\begin{equation}
    \mM{proj} = \begin{pmatrix}
        \dfrac{1}{a \tan(\text{FOV}_y/2)} & 0 & 0 & 0 \\
        0 & \dfrac{1}{\tan(\text{FOV}_y/2)} & 0 & 0 \\
        0 & 0 & -\dfrac{f+n}{f-n} & -\dfrac{2fn}{f-n} \\
        0 & 0 & -1 & 0
    \end{pmatrix}
\end{equation}

\noindent
\textbf{Note:} This is an OpenGL-style projection ($Z_{clip} \in [-1,1]$). FIR provides a Vulkan-native variant with $Z_{clip} \in [0,1]$ but it is unused.

%-----------------------------------------------------------------------------
\subsection{Matrix Upload Convention}
%-----------------------------------------------------------------------------

File: \texttt{src/Graphics/Haskan/Engine.hs:591--592}

The \texttt{linear} package stores matrices in \textbf{row-major} order. The GPU (FIR/SPIR-V) expects \textbf{column-major}. An explicit transpose is performed before upload:
\begin{align}
    \mM{view}^{GPU}  &= (\mM{view}^{CPU})^\top \\
    \mM{proj}^{GPU}  &= (\mM{proj}^{CPU})^\top \\
    \mM{model}^{GPU} &= (\mM{model}^{CPU})^\top
\end{align}

\subsection{Vertex Transformations}
%-----------------------------------------------------------------------------

\begin{align}
    \vv{p}_{world} &= \mM{model}_i \cdot \begin{pmatrix} p_x \\ p_y \\ p_z \\ 1 \end{pmatrix} \\
    \vv{n}_{world} &= \mM{model}_i \cdot \begin{pmatrix} n_x \\ n_y \\ n_z \\ 0 \end{pmatrix} \\
    \vv{t}_{world} &= \mM{model}_i \cdot \begin{pmatrix} t_x \\ t_y \\ t_z \\ w \end{pmatrix} \\

\noindent
\textbf{Bug:} Normals and tangents are transformed by the raw model matrix. For non-uniform scale, the correct transform is:
\begin{equation}
    \vv{n}_{world} = (\mM{model}^{-1})^\top \cdot \vv{n}
    \quad \text{(inverse-transpose of upper-left 3$\times$3)}
\end{equation}

\section{Tangent Space Computation (CPU)}
%=============================================================================

File: \texttt{src/Graphics/Haskan/Utils/TangentSpace.hs:16--106}

%-----------------------------------------------------------------------------
\subsection{Per-Triangle Tangent/Bitangent}
%-----------------------------------------------------------------------------

For a triangle with vertices $\vv{p}_0, \vv{p}_1, \vv{p}_2$ and UVs $(u_0,v_0), (u_1,v_1), (u_2,v_2)$:
\begin{align}
    \vv{e}_1 &= \vv{p}_1 - \vv{p}_0, \quad \vv{e}_2 = \vv{p}_2 - \vv{p}_0 \\
    \Delta u_1 &= u_1 - u_0, \quad \Delta v_1 = v_1 - v_0 \\
    \Delta u_2 &= u_2 - u_0, \quad \Delta v_2 = v_2 - v_0 \\
    \det &= \Delta u_1 \Delta v_2 - \Delta u_2 \Delta v_1
\end{align}

If $|\det| < 10^{-6}$ (degenerate UVs), fallback to $\vv{T} = (1,0,0)$, $\vv{B} = (0,1,0)$. Otherwise:
\begin{align}
    f &= \frac{1}{\det} \\
    T_x &= f(\Delta v_2 \cdot e_{1x} - \Delta v_1 \cdot e_{2x}) \\
    T_y &= f(\Delta v_2 \cdot e_{1y} - \Delta v_1 \cdot e_{2y}) \\
    T_z &= f(\Delta v_2 \cdot e_{1z} - \Delta v_1 \cdot e_{2z}) \\
    B_x &= f(-\Delta u_2 \cdot e_{1x} + \Delta u_1 \cdot e_{2x}) \\
    B_y &= f(-\Delta u_2 \cdot e_{1y} + \Delta u_1 \cdot e_{2y}) \\
    B_z &= f(-\Delta u_2 \cdot e_{1z} + \Delta u_1 \cdot e_{2z})
\end{align}

Tangents/bitangents are accumulated per-vertex (sum over adjacent triangles).

%-----------------------------------------------------------------------------
\subsection{Per-Vertex Finalization}
%-----------------------------------------------------------------------------

Given accumulated $\vv{T}_{sum}$, $\vv{B}_{sum}$, and vertex normal $\vv{n}$:
\begin{align}
    t_{\text{dot}n} &= \vv{T}_{sum} \cdot \vv{n} \\
    \vv{T}_{\perp} &= \vv{T}_{sum} - t_{\text{dot}n} \cdot \vv{n} \\
    \vv{T}_{\text{final}} &= \normalize(\vv{T}_{\perp}) \\
    \vv{B}_{\text{ref}} &= \vv{n} \times \vv{T}_{\text{final}} \\
    w &= \begin{cases}
        +1 & \text{if } \vv{B}_{sum} \cdot \vv{B}_{\text{ref}} > 0 \\
        -1 & \text{otherwise}
    \end{cases}
\end{align}

Output tangent: $\vv{t}_h = (T_x, T_y, T_z, w)$

%-----------------------------------------------------------------------------
\subsection{Metallic/Roughness}
%-----------------------------------------------------------------------------

\begin{align}
    \text{useMrTex} &= [\text{mrIdx} \neq 0] \\
    \vv{c}_{mr} &= \text{sample}(\text{tex}, \text{mrIdx}, \vv{uv}) \\
    \text{metallic} &= \begin{cases}
        c_{mr}.b & \text{if useMrTex} \\
        \text{metallicFactor} & \text{otherwise}
    \end{cases} \\
    \text{roughness} &= \begin{cases}
        c_{mr}.g & \text{if useMrTex} \\
        \text{roughnessFactor} & \text{otherwise}
    \end{cases}
\end{align}

%-----------------------------------------------------------------------------
\subsection{TBN Construction}
%-----------------------------------------------------------------------------

\begin{align}
    \vv{N} &= \normalize((n_x, n_y, n_z)) \\
    \vv{T} &= \normalize((t_x, t_y, t_z)) \\
    \vv{B}_{raw} &= \vv{N} \times \vv{T} \\
    \vv{B} &= w \cdot \vv{B}_{raw}
\end{align}

No Gram-Schmidt re-orthogonalization in the shader.

%-----------------------------------------------------------------------------
\subsection{Normal Mapping}
%-----------------------------------------------------------------------------

\begin{align}
    \text{useNormalTex} &= [\text{normalIdx} \neq 0] \\
    \vv{n}_{raw} &= \text{sample}(\text{tex}, \text{normalIdx}, \vv{uv}) \\
    \vv{n}_{tan} &= 2 \cdot \vv{n}_{raw} - (1, 1, 1) \quad \text{(decode [0,1] to [-1,1])} \\
    \vv{N}_{world} &= \begin{cases}
        n_x \cdot \vv{T} + n_y \cdot \vv{B} + n_z \cdot \vv{N} & \text{if useNormalTex} \\
        \vv{N} & \text{otherwise}
    \end{cases}
\end{align}

%-----------------------------------------------------------------------------
\subsection{Occlusion}
%-----------------------------------------------------------------------------

\begin{align}
    \text{useOcclTex} &= [\text{occlusionIdx} \neq 0] \\
    o_{raw} &= \text{sample}(\text{tex}, \text{occlusionIdx}, \vv{uv}).r \\
    \text{AO} &= \begin{cases}
        1.0 - \text{occlusionStrength} \cdot (1.0 - o_{raw}) & \text{if useOcclTex} \\
        1.0 & \text{otherwise}
    \end{cases}
\end{align}

%-----------------------------------------------------------------------------
\subsection{Emissive}
%-----------------------------------------------------------------------------

\begin{align}
    \text{useEmissiveTex} &= [\text{emissiveIdx} \neq 0] \\
    \vv{e} &= \begin{cases}
        \text{sample}(\text{tex}, \text{emissiveIdx}, \vv{uv}).rgb & \text{if useEmissiveTex} \\
        (0, 0, 0) & \text{otherwise}
    \end{cases}
\end{align}

%-----------------------------------------------------------------------------
\subsection{G-Buffer Output Encoding}
%-----------------------------------------------------------------------------

\begin{align}
    \text{out\_position} &= (p_{wx}, p_{wy}, p_{wz}, \text{metallic}) \\
    \text{out\_normal}   &= \left(\frac{N_x}{2} + 0.5, \frac{N_y}{2} + 0.5, \frac{N_z}{2} + 0.5, \text{roughness}\right) \\
    \text{out\_albedo}   &= (c_r, c_g, c_b, \text{AO}) \\
    \text{out\_emissive} &= (e_r, e_g, e_b, 1.0)
\end{align}

Normal is encoded from $[-1,1]$ to $[0,1]$ for \texttt{RGBA8\_UNORM} storage.

%-----------------------------------------------------------------------------
\subsection{Fullscreen Triangle Vertex Generation}
%-----------------------------------------------------------------------------

No vertex input. Positions and UVs generated from \texttt{gl\_VertexIndex}:
\begin{align}
    i &= \text{gl\_VertexIndex} \in \{0, 1, 2\} \\
    (x, y) &= \begin{cases}
        (-1, -1) & i = 0 \\
        (3, -1)  & i = 1 \\
        (-1, 3)  & i = 2
    \end{cases} \\
    (u, v) &= \begin{cases}
        (0, 1)  & i = 0 \\
        (2, 1)  & i = 1 \\
        (0, -1) & i = 2
    \end{cases}
\end{align}

The triangle $(-1,-1), (3,-1), (-1,3)$ covers the entire clip-space viewport. UV $v$ is flipped so that screen top maps to texture top (Vulkan convention).

%-----------------------------------------------------------------------------
\subsection{G-Buffer Decoding}
%-----------------------------------------------------------------------------

Sample at interpolated UV $\vv{uv}$:
\begin{align}
    (p_x, p_y, p_z, \text{metallic}) &= \text{sample}(\text{gbuf\_position}, \vv{uv}) \\
    (n'_x, n'_y, n'_z, \text{roughness}) &= \text{sample}(\text{gbuf\_normal}, \vv{uv}) \\
    (a_r, a_g, a_b, \text{AO}) &= \text{sample}(\text{gbuf\_albedo}, \vv{uv}) \\
    (e_r, e_g, e_b, \_) &= \text{sample}(\text{gbuf\_emissive}, \vv{uv})
\end{align}

Normal decode (UNORM $[0,1]$ back to $[-1,1]$):
\begin{align}
    n_x &= 2 n'_x - 1 \\
    n_y &= 2 n'_y - 1 \\
    n_z &= 2 n'_z - 1 \\
    \vv{N} &= \normalize((n_x, n_y, n_z))
\end{align}

%-----------------------------------------------------------------------------
\subsection{View Vector}
%-----------------------------------------------------------------------------

Camera position $\vv{c} = (c_x, c_y, c_z)$ from push constant:
\begin{align}
    \vv{V}_{raw} &= \vv{c} - \vv{p}_{world} \\
    \vv{V} &= \normalize(\vv{V}_{raw})
\end{align}

%-----------------------------------------------------------------------------
\subsection{Directional Light}
%-----------------------------------------------------------------------------

Hardcoded light direction (normalized):
\begin{equation}
    \vv{L} = \frac{1}{\sqrt{3}}(1, 1, 1)
\end{equation}

Dot products:
\begin{align}
    N \cdot L &= \max(0, \vv{N} \cdot \vv{L}) \\
    N \cdot V &= \max(0, \vv{N} \cdot \vv{V})
\end{align}

Half vector:
\begin{align}
    \vv{H}_{raw} &= \vv{L} + \vv{V} \\
    \vv{H} &= \normalize(\vv{H}_{raw}) \\
    N \cdot H &= \max(0, \vv{N} \cdot \vv{H}) \\
    V \cdot H &= \max(0, \vv{V} \cdot \vv{H})
\end{align}

%-----------------------------------------------------------------------------
\subsection{PBR BRDF (Direct Lighting)}
%-----------------------------------------------------------------------------

Fresnel reflectance at normal incidence (Schlick approximation base):
\begin{equation}
    \vv{F}_0 = 0.04 \cdot (1 - \text{metallic}) + \vv{a}_{rgb} \cdot \text{metallic}
\end{equation}

Fresnel term (Schlick):
\begin{equation}
    \vv{F} = \vv{F}_0 + (1 - \vv{F}_0) \cdot (1 - V \cdot H)^5
\end{equation}

GGX Normal Distribution Function (Trowbridge-Reitz):
\begin{align}
    \alpha &= \text{roughness}^2 \\
    D &= \frac{\alpha^2}{\pi \cdot \left((N \cdot H)^2 (\alpha^2 - 1) + 1\right)^2}
\end{align}

Geometry term (Schlick-Smith approximation of Smith GGX):
\begin{align}
    k &= \frac{(\alpha + 1)^2}{8} \\
    G_1(L) &= \frac{N \cdot L}{N \cdot L \cdot (1 - k) + k} \\
    G_1(V) &= \frac{N \cdot V}{N \cdot V \cdot (1 - k) + k} \\
    G &= G_1(L) \cdot G_1(V)
\end{align}

Specular BRDF:
\begin{equation}
    \vv{f}_{spec} = \frac{D \cdot \vv{F} \cdot G}{4 \cdot (N \cdot L) \cdot (N \cdot V) + 0.001}
\end{equation}

Diffuse BRDF (Lambert):
\begin{equation}
    \vv{f}_{diff} = \frac{\vv{a}_{rgb} \cdot (1 - \text{metallic})}{\pi}
\end{equation}

Combined direct lighting:
\begin{equation}
    \vv{L}_{dir} = (\vv{f}_{diff} + \vv{f}_{spec}) \cdot (N \cdot L)
\end{equation}

%-----------------------------------------------------------------------------
\subsection{Image-Based Lighting (IBL)}
%-----------------------------------------------------------------------------

Reflection vector:
\begin{align}
    R_{dotN} &= 2 (\vv{V} \cdot \vv{N}) \\
    \vv{R} &= R_{dotN} \cdot \vv{N} - \vv{V}
\end{align}

Fresnel for IBL (uses $N \cdot V$ instead of $V \cdot H$ since there is no half-vector):
\begin{equation}
    \vv{F}_{IBL} = \vv{F}_0 + (1 - \vv{F}_0) \cdot (1 - N \cdot V)^5
\end{equation}

BRDF LUT sampling (split-sum approximation):
\begin{equation}
    (scale, bias) = \text{sample}(\text{brdf\_lut}, (N \cdot V, \text{roughness}))
\end{equation}

Irradiance (diffuse IBL):
\begin{equation}
    \vv{L}_{irr} = \text{sample}(\text{irradiance\_map}, \vv{N}) \cdot \vv{a}_{rgb} \cdot (1 - \text{metallic}) \cdot \text{AO} \cdot 0.3
\end{equation}

Radiance (specular IBL):
\begin{align}
    lod &= \text{roughness} \cdot 9.0 \quad \text{(9 mip levels on 512px cubemap)} \\
    \vv{L}_{rad} &= \text{sample}(\text{env\_map}, \vv{R}, lod) \cdot (\vv{F}_{IBL} \cdot scale + bias) \cdot 0.3
\end{align}

%-----------------------------------------------------------------------------
\subsection{Final Color Composition}
%-----------------------------------------------------------------------------

\begin{equation}
    \vv{C}_{hdr} = \vv{L}_{dir} + \vv{e} + \vv{L}_{irr} + \vv{L}_{rad}
\end{equation}

Tone mapping (Reinhard):
\begin{equation}
    \vv{C}_{tm} = \frac{\vv{C}_{hdr}}{\vv{C}_{hdr} + 1}
\end{equation}

Gamma correction (approximate $\gamma = 2.0$):
\begin{equation}
    \vv{C}_{final} = \sqrt{\vv{C}_{tm}}
\end{equation}

\section{Procedural Mesh Formulas}
%=============================================================================

File: \texttt{src/Graphics/Haskan/Mesh.hs}

%-----------------------------------------------------------------------------
\subsection{UV Sphere}
%-----------------------------------------------------------------------------

Parameters: latitude segments $N_{lat}$, longitude segments $N_{lon}$, radius $r$.

For $i \in [0, N_{lat}]$, $j \in [0, N_{lon}]$:
\begin{align}
    \theta &= \pi \cdot \frac{i}{N_{lat}} \quad \text{(polar angle, 0 at north pole)} \\
    \phi   &= 2\pi \cdot \frac{j}{N_{lon}} \quad \text{(azimuthal angle)} \\
    p_x &= r \sin\theta \cos\phi \\
    p_y &= r \cos\theta \\
    p_z &= r \sin\theta \sin\phi \\
    \vv{N} &= \normalize((p_x, p_y, p_z)) \\
    u &= 1.0 - \frac{j}{N_{lon}} \quad \text{(flipped to prevent mirrored text)} \\
    v &= \frac{i}{N_{lat}} \\
    T_x &= \sin\phi \\
    T_y &= 0 \\
    T_z &= -\cos\phi \\
    \vv{t}_h &= (T_x, T_y, T_z, 1)
\end{align}

%-----------------------------------------------------------------------------
\subsection{Unit Cube}
%-----------------------------------------------------------------------------

Centered at origin, side length 1, half-extent $s = 0.5$. 24 vertices (4 per face $\times$ 6 faces).

Front face ($+Z$, normal $(0,0,1)$):
\begin{align}
    \vv{p}_0 &= (-s, -s, +s), \quad \vv{uv}_0 = (0, 1) \\
    \vv{p}_1 &= (+s, -s, +s), \quad \vv{uv}_1 = (1, 1) \\
    \vv{p}_2 &= (+s, +s, +s), \quad \vv{uv}_2 = (1, 0) \\
    \vv{p}_3 &= (-s, +s, +s), \quad \vv{uv}_3 = (0, 0) \\
    \vv{t}_h &= (1, 0, 0, 1)
\end{align}

UV convention: $(0,0)$ = top-left (Vulkan). Winding: front/back/right use $(0,2,1,0,3,2)$; left/top/bottom use $(0,1,2,0,2,3)$.

\section{BRDF LUT Generation (CPU)}
%=============================================================================

File: \texttt{src/Graphics/Haskan/Vulkan/BRDF.hs}

%-----------------------------------------------------------------------------
\subsection{Integration Setup}
%-----------------------------------------------------------------------------

For each texel at integer coordinates $(x, y)$ in a $W \times H$ LUT:
\begin{align}
    u &= \frac{x + 0.5}{W} \quad (= N \cdot V) \\
    v &= \frac{y + 0.5}{H} \quad (= \text{roughness})
\end{align}

Integration with $N = 128$ Monte Carlo samples:
\begin{align}
    (scale, bias) &= \text{integrateBRDF}(u, v, N) \\
    \text{pixel} &= (\clamp_{01}(scale), \clamp_{01}(bias), 0, 1) \cdot 255
\end{align}

%-----------------------------------------------------------------------------
\subsection{Monte Carlo Integration}
%-----------------------------------------------------------------------------

Given $N \cdot V = \cos\theta_v$ and roughness:
\begin{align}
    \vv{n} &= (0, 0, 1) \\
    \vv{v} &= (\sqrt{1 - (N \cdot V)^2}, 0, N \cdot V) \\
    \alpha &= \text{roughness}^2
\end{align}

For each sample with randoms $(r_1, r_2)$:
\begin{align}
    \phi &= 2\pi r_1 \\
    \cos\theta_h &= \sqrt{\frac{1 - r_2}{1 + (\alpha^2 - 1) r_2}} \\
    \sin\theta_h &= \sqrt{1 - \cos^2\theta_h} \\
    \vv{h} &= \normalize((\sin\theta_h \cos\phi, \sin\theta_h \sin\phi, \cos\theta_h)) \\
    \vv{l} &= \normalize(2 (\vv{v} \cdot \vv{h}) \vv{h} - \vv{v}) \\
    N \cdot l &= \max(0, \vv{n} \cdot \vv{l}) \\
    N \cdot h &= \max(0, \vv{n} \cdot \vv{h}) \\
    V \cdot h &= \max(0, \vv{v} \cdot \vv{h})
\end{align}

GGX distribution:
\begin{equation}
    D = \frac{\alpha^2}{\pi \left((N \cdot h)^2 (\alpha^2 - 1) + 1\right)^2}
\end{equation}

Geometry:
\begin{align}
    k &= \frac{(\alpha + 1)^2}{8} \\
    G_1(v) &= \frac{N \cdot V}{N \cdot V \cdot (1 - k) + k} \\
    G_1(l) &= \frac{N \cdot l}{N \cdot l \cdot (1 - k) + k} \\
    G &= G_1(v) \cdot G_1(l)
\end{align}

Visibility term (already cancels PDF in importance sampling):
\begin{equation}
    V_{is} = \begin{cases}
        \dfrac{G \cdot (V \cdot h)}{(N \cdot h) \cdot (N \cdot V)} & \text{if } N \cdot l > 0 \text{ and } N \cdot V > 0 \\
        0 & \text{otherwise}
    \end{cases}
\end{equation}

Fresnel:
\begin{equation}
    F_c = (1 - V \cdot h)^5
\end{equation}

Accumulation:
\begin{align}
    scale' &= V_{is} \cdot (1 - F_c) \\
    bias'  &= V_{is} \cdot F_c
\end{align}

Final averaging:
\begin{equation}
    scale = \frac{1}{N} \sum scale', \quad bias = \frac{1}{N} \sum bias'
\end{equation}

\section{Coordinate Transform Summary}
%=============================================================================

The full vertex transformation chain:
\begin{equation}
    \vv{p}_{clip} = \mM{P} \cdot \mM{V} \cdot \mM{M} \cdot \vv{p}_{local}
\end{equation}

Where:
\begin{align}
    \mM{M} &= \mM{T} \cdot \mM{R} \cdot \mM{S} \quad \text{(model matrix)} \\
    \mM{V} &= \text{lookAt}(eye, center, up) \quad \text{(view matrix)} \\
    \mM{P} &= \text{perspective}(fovy, aspect, near, far) \quad \text{(projection matrix)}
\end{align}

Pipeline spaces:
\begin{equation}
    \vv{p}_{local} \xrightarrow{\mM{M}} \vv{p}_{world} \xrightarrow{\mM{V}} \vv{p}_{view} \xrightarrow{\mM{P}} \vv{p}_{clip} \xrightarrow{/w} \vv{p}_{NDC} \xrightarrow{\text{viewport}} \vv{p}_{screen}
\end{equation}

Viewport transform (Vulkan):
\begin{align}
    x_{screen} &= \frac{x_{NDC} + 1}{2} \cdot width \\
    y_{screen} &= \frac{1 - y_{NDC}}{2} \cdot height \\
    z_{depth}  &= z_{NDC} \quad \text{(with depth range [0,1])}
\end{align}

Note: Because the projection produces $Y_{NDC}$ up but Vulkan expects $Y_{NDC}$ down, the effective viewport $Y$ is inverted. The engine uses clockwise front-face winding to compensate.
